EXPLAIN ANALYZE
SELECT
    s.student_id,
    s.last_name,
    s.first_name,
    c.title,
    ex.attempt_no,
    ex.exam_date,
    ex.grade
FROM "Exam" ex
JOIN "Enrollment" en
  ON en.student_id    = ex.student_id
 AND en.course_id     = ex.course_id
 AND en.instructor_id = ex.instructor_id
JOIN "Student" s
  ON s.student_id = ex.student_id
JOIN "Course" c
  ON c.course_id = ex.course_id
WHERE s.group_code = 'KP-21'
  AND ex.grade BETWEEN 60 AND 90
  AND c.title ILIKE '%Data%'
  AND ex.exam_date BETWEEN DATE '2022-01-01' AND DATE '2025-12-31'
ORDER BY s.last_name, s.first_name, ex.exam_date DESC;
