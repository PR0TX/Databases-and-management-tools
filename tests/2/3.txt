EXPLAIN ANALYZE
SELECT
    s.student_id,
    s.last_name,
    s.first_name,
    COUNT(*)               AS exam_count,
    ROUND(AVG(ex.grade),1) AS avg_grade
FROM "Exam" ex
JOIN "Student" s
  ON s.student_id = ex.student_id
WHERE s.group_code = 'KP-21'
GROUP BY s.student_id, s.last_name, s.first_name
HAVING COUNT(*) >= 3
ORDER BY avg_grade DESC;
