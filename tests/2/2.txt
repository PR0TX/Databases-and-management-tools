EXPLAIN ANALYZE
SELECT
    c.title,
    (i.last_name || ' ' || i.first_name) AS instructor,
    COUNT(*)               AS attempts,
    ROUND(AVG(ex.grade),1) AS avg_grade
FROM "Exam" ex
JOIN "Enrollment" en
  ON en.student_id    = ex.student_id
 AND en.course_id     = ex.course_id
 AND en.instructor_id = ex.instructor_id
JOIN "Course" c
  ON c.course_id = ex.course_id
JOIN "Instructor" i
  ON i.instructor_id = ex.instructor_id
WHERE ex.exam_date BETWEEN DATE '2022-01-01' AND DATE '2025-12-31'
  AND c.title ILIKE '%Data%'
GROUP BY c.title, i.last_name, i.first_name
HAVING COUNT(*) >= 50
ORDER BY avg_grade DESC;
